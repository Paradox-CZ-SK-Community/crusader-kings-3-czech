## Supported scopes: Culture, Faith, Landed Title
##
## _yy   = -ý
## _yych = -ých
## _eeho = -ého
## _eemu = -ému
## _eem  = -ém
## _yym  = -ým
## _yymi = -ými
## _aa   = -á
## _ou   = -ou
## _sti  = -ští / -čtí / -í
## _o    = -o
## _u    = -u
## _ee   = -é

	type = all

	text = {
		localization_key = CZ_Adjective
