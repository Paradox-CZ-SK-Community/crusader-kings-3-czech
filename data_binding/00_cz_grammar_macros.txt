##################################################
# CZECH GRAMMAR MACROS
##################################################

##################################################
# TITLE TOKENIZER
#
# CZ_TitleTokenChars is defined in:
# localization/polish/00_cz_custom_loc_l_polish.yml
#
# Space is intentionally NOT included in the
# character set and therefore works as separator.
#
# Current supported title length: 1-3 words.
##################################################


# ------------------------------------------------
# DETECTION
# ------------------------------------------------

macro = {
	description = "True if TITLE contains at least two words"
	definition = "CZTitleHasSecond(TITLE)"
	replace_with = "Not(StringIsEmpty(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))))"
}

macro = {
	description = "True if TITLE contains at least three words"
	definition = "CZTitleHasThird(TITLE)"
	replace_with = "Not(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))))"
}


# ------------------------------------------------
# BASIC STRING HELPERS
# ------------------------------------------------

macro = {
	description = "Remove the first word from TITLE"
	definition = "CZTitleDropFirst(TITLE)"
	replace_with = "TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))"
}

macro = {
	description = "Remove the last word from TITLE"
	definition = "CZTitleDropLast(TITLE)"
	replace_with = "TrimBack(TrimBackChars(TITLE, Localize('CZ_TitleTokenChars')))"
}


# ------------------------------------------------
# WORD 1
# ------------------------------------------------

macro = {
	description = "Return first word of a one-to-three-word TITLE"
	definition = "CZTitleWord1(TITLE)"
	replace_with = "Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))), TITLE, Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))), TrimBack(TrimBackChars(TITLE, Localize('CZ_TitleTokenChars'))), TrimBack(TrimBackChars(TrimBack(TrimBackChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))))"
}


# ------------------------------------------------
# WORD 2
# ------------------------------------------------

macro = {
	description = "Return second word of a one-to-three-word TITLE"
	definition = "CZTitleWord2(TITLE)"
	replace_with = "Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))), '', Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))), TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), TrimBack(TrimBackChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))))"
}


# ------------------------------------------------
# WORD 3
# ------------------------------------------------

macro = {
	description = "Return third word of a one-to-three-word TITLE"
	definition = "CZTitleWord3(TITLE)"
	replace_with = "Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))), '', TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars'))))"
}

# GENITIVE

macro = {
	description = "Czech title word genitive (male)"
	definition = "CZTitleWord_G_M(WORD)"
	replace_with = "Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'ý'))),Concatenate(TrimBackChars(WORD,'ý'),'ého'),Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'í'))),Concatenate(TrimBackChars(WORD,'í'),'ího'),Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'a'))),Concatenate(TrimBackChars(WORD,'a'),'y'),Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'o'))),Concatenate(TrimBackChars(WORD,'o'),'a'),Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'ě'))),Concatenate(WORD,'te'),Select_CString(And(Not(EqualTo_string(WORD,TrimBackChars(WORD,'e'))),Not(EqualTo_string(TrimBackChars(WORD,'e'),TrimBackChars(TrimBackChars(WORD,'e'),'ž')))),Concatenate(WORD,'te'),Select_CString(And(Not(EqualTo_string(WORD,TrimBackChars(WORD,'e'))),Not(EqualTo_string(TrimBackChars(WORD,'e'),TrimBackChars(TrimBackChars(WORD,'e'),'c')))),WORD,Select_CString(Or(Or(EqualTo_string(WORD,'Král'),EqualTo_string(WORD,'král')),Or(EqualTo_string(WORD,'Velkokrál'),EqualTo_string(WORD,'Velekrál'))),Concatenate(TrimBackChars(WORD,'l'),'le'),Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'bcčdďfghjklmnňpqrřsštťvwxzž'))),Select_CString(Not(EqualTo_string(WORD,TrimBackChars(WORD,'cčďjňřsšťxzž'))),Concatenate(WORD,'e'),Concatenate(WORD,'a')),WORD)))))))))"
}

}
