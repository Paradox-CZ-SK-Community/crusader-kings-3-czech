# ------------------------------------------------
# JOIN
# ------------------------------------------------

macro = {
	description = "Join two non-empty strings with one space"
	definition = "CZJoin(A,B)"
	replace_with = "Select_CString(StringIsEmpty(A),B,Select_CString(StringIsEmpty(B),A,Concatenate(A,Concatenate(' ',B))))"
}

# ------------------------------------------------
# CHARACTER NAME PREFIX
# ------------------------------------------------

# Exact reconstruction is intentional: CK3 character names may
# contain NBSP, e.g. "Král Sancho II.". The helper must not
# tokenize that prefix by ordinary spaces.

macro = {
	description = "True if PREFIX is an exact CK3 character-name prefix of TEXT"
	definition = "CZHasNamePrefix(TEXT,PREFIX)"
	replace_with = "And(Not(StringIsEmpty(PREFIX)),EqualTo_string(TEXT,Concatenate(PREFIX,Select_CString(EqualTo_string(TEXT,PREFIX),'',Concatenate(' ',TrimFrontChars(TEXT,PREFIX))))))"
}

macro = {
	description = "Replace an exact CK3 character-name prefix while preserving the remaining suffix"
	definition = "CZReplaceNamePrefix(TEXT,PREFIX,REPLACEMENT)"
	replace_with = "Select_CString(CZHasNamePrefix(TEXT,PREFIX),CZJoin(REPLACEMENT,TrimFrontChars(TEXT,PREFIX)),TEXT)"
}


##################################################
# TITLE TOKENIZER
##################################################
# CZ_TitleTokenChars is defined in:
# localization/polish/custom_localization/cz_custom_loc/00_cz_custom_loc_l_polish.yml
#
# Space is intentionally NOT included in CZ_TitleTokenChars and therefore
# acts as the separator. Current supported title length: 1-3 words.

# ------------------------------------------------
# DETECTION
# ------------------------------------------------

macro = {
	description = "True if TITLE contains at least two words"
	definition = "CZTitleHasSecond(TITLE)"
	replace_with = "Not(StringIsEmpty(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))))"
}

macro = {
	description = "True if TITLE contains at least three words"
	definition = "CZTitleHasThird(TITLE)"
	replace_with = "Not(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))))"
}

# ------------------------------------------------
# WORD EXTRACTION
# ------------------------------------------------

macro = {
	description = "Return first word of a one-to-three-word TITLE"
	definition = "CZTitleWord1(TITLE)"
	replace_with = "Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))), TITLE, Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))), TrimBack(TrimBackChars(TITLE, Localize('CZ_TitleTokenChars'))), TrimBack(TrimBackChars(TrimBack(TrimBackChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))))"
}

macro = {
	description = "Return second word of a one-to-three-word TITLE"
	definition = "CZTitleWord2(TITLE)"
	replace_with = "Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars')))), '', Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))), TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), TrimBack(TrimBackChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))))"
}

macro = {
	description = "Return third word of a one-to-three-word TITLE"
	definition = "CZTitleWord3(TITLE)"
	replace_with = "Select_CString(StringIsEmpty(TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars')))), '', TrimFront(TrimFrontChars(TrimFront(TrimFrontChars(TITLE, Localize('CZ_TitleTokenChars'))), Localize('CZ_TitleTokenChars'))))"
}
